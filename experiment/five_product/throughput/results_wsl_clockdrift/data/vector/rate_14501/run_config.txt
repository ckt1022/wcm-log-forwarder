# Vector：VRL remap 實作與其他產品相同的 parse／filter／format。
sources:
  tcp_in:
    type: socket
    mode: tcp
    address: "0.0.0.0:15140"
    decoding:
      codec: json

transforms:
  latency_filter:
    type: remap
    inputs: ["tcp_in"]
    # rank < warn 的 event 以 abort 丟棄
    drop_on_abort: true
    drop_on_error: false
    source: |
      # ── 邏輯對齊 latency_filter.lua / filter_latency_test.rb ──
      # parse : ts 原字串搬到 timestamp、level 字串/數值映射、msg → message、
      #         根層非保留欄位與 att.* 攤平成 tags
      # filter: 只保留 level >= warn（MIN_KEEP_RANK = 2）
      # format: json-flat 欄位集合 id/timestamp/level/message/lang/<tags>
      # 關鍵約束：timestamp 必須是 gen 原始 ts 字串原封不動，
      # HTTP 接收端（common/tools/httpsink）靠這個欄位計算延遲。

      # level → rank（字串不分大小寫；數值 0~6 取整；缺失/未知 → info=1）
      rank = 1
      if is_string(.level) {
        r = get!({"debug": 0, "info": 1, "warn": 2, "error": 3,
                  "crit": 4, "alert": 5, "emerg": 6},
                 [downcase(string!(.level))])
        if r != null { rank = int!(r) }
      } else if is_integer(.level) || is_float(.level) {
        n = to_int(floor(to_float!(.level)))
        if n >= 0 && n <= 6 { rank = n }
      }

      # filter：drop debug / info
      if rank < 2 { abort }

      level_name = string!(get!(["debug", "info", "warn", "error",
                                 "crit", "alert", "emerg"], [rank]))

      ts_val = ""
      if is_string(.ts) { ts_val = string!(.ts) }
      msg_val = ""
      if is_string(.msg) { msg_val = string!(.msg) }

      out = {
        "id":        0,
        "timestamp": ts_val,
        "level":     level_name,
        "message":   msg_val,
        "lang":      "VRL"
      }

      # 根層非保留欄位攤平成 tags（對應 collect_all_tags 的第一段）。
      # 除 gen record 的保留欄位外，還須排除 vector 對 event 注入的欄位
      # （socket source：host/port/source_type；legacy namespace：timestamp）
      for_each(object(.)) -> |k, v| {
        if k != "ts" && k != "level" && k != "msg" && k != "att" &&
           k != "timestamp" && k != "source_type" && k != "host" && k != "port" {
          sv = ""
          if is_string(v) {
            sv = string!(v)
          } else if is_object(v) || is_array(v) {
            sv = encode_json(v)
          } else {
            sv = to_string!(v)
          }
          out = set!(out, [k], sv)
        }
      }

      # att 物件內欄位攤平（對應 collect_all_tags 的第二段）
      if is_object(.att) {
        for_each(object!(.att)) -> |k, v| {
          sv = ""
          if is_string(v) {
            sv = string!(v)
          } else if is_object(v) || is_array(v) {
            sv = encode_json(v)
          } else {
            sv = to_string!(v)
          }
          out = set!(out, [k], sv)
        }
      }

      . = out
